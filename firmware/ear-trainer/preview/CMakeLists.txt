cmake_minimum_required(VERSION 3.16)
project(ear_preview C CXX ASM)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(ear_preview preview.c ../main/ear_model.c ../main/ear_controller.c ../main/ear_ui.c ../main/ear_font_16.c)
target_include_directories(ear_preview PRIVATE ../main)
target_link_libraries(ear_preview PRIVATE lvgl m)
target_compile_options(ear_preview PRIVATE -Wall -Wextra -Werror)
