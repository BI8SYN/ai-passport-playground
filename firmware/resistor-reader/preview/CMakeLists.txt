cmake_minimum_required(VERSION 3.16)
project(resistor_preview C CXX ASM)
set(RR_TEXTURE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/resistor/resistor_textures.bin")
configure_file(textures.S.in textures.S @ONLY)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/textures.S PROPERTIES OBJECT_DEPENDS "${RR_TEXTURE_FILE}")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(resistor_preview preview.c ../main/resistor_model.c ../main/resistor_ui.c ../main/resistor_render.c ../main/resistor_layout.c ../main/resistor_input.c ../../../tools/renderer/resistor_art.c ${CMAKE_CURRENT_BINARY_DIR}/textures.S ../main/resistor_font_16.c)
target_include_directories(resistor_preview PRIVATE ../main)
target_link_libraries(resistor_preview PRIVATE lvgl m)
target_compile_options(resistor_preview PRIVATE -Wall -Wextra -Werror)
