cmake_minimum_required(VERSION 3.16)
project(resistor_preview C CXX)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl)
add_executable(resistor_preview preview.c ../main/resistor_model.c ../main/resistor_ui.c ../main/resistor_font_16.c)
target_include_directories(resistor_preview PRIVATE ../main)
target_link_libraries(resistor_preview PRIVATE lvgl m)
target_compile_options(resistor_preview PRIVATE -Wall -Wextra -Werror)
